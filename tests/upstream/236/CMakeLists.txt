cmake_minimum_required(VERSION 3.16)
project(upstream236 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTOUIC_SEARCH_PATHS "${CMAKE_CURRENT_LIST_DIR}")
set(NULLOY_QT_MAJOR 6 CACHE STRING "Qt major version (5 or 6)")
find_package(Qt${NULLOY_QT_MAJOR} REQUIRED COMPONENTS Widgets Test)
set(N_SRC "${CMAKE_CURRENT_LIST_DIR}/../../../src" CACHE PATH "Production source directory")
add_executable(testWindowRestoration
    testWindowRestoration.cpp form.ui
    ${N_SRC}/mainWindow.cpp ${N_SRC}/mainWindow.h
    ${N_SRC}/settings.cpp ${N_SRC}/settings.h)
target_include_directories(testWindowRestoration PRIVATE ${N_SRC} ${N_SRC}/platform)
target_compile_definitions(testWindowRestoration PRIVATE _N_NO_SKINS_)
target_link_libraries(testWindowRestoration PRIVATE
    Qt${NULLOY_QT_MAJOR}::Widgets Qt${NULLOY_QT_MAJOR}::Test)
if(WIN32)
    target_sources(testWindowRestoration PRIVATE ${N_SRC}/platform/w7TaskBar.cpp ${N_SRC}/platform/w7TaskBar.h)
    target_compile_definitions(testWindowRestoration PRIVATE WIN32)
    target_link_libraries(testWindowRestoration PRIVATE dwmapi ole32 uuid)
    if(NULLOY_QT_MAJOR STREQUAL "5")
        find_package(Qt5 REQUIRED COMPONENTS WinExtras)
        target_link_libraries(testWindowRestoration PRIVATE Qt5::WinExtras)
    endif()
endif()
enable_testing()
add_test(NAME upstream236 COMMAND testWindowRestoration)
set_tests_properties(upstream236 PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
